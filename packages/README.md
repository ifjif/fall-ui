# 使用

    npm i fall-ui

    //组件导入
    import FallUI from 'fall-ui'
    //icon导入
    import FallIcon from 'fall-ui/icons'
    //样式引入
    import "fall-ui/themes/index.css"
    //单个导入
    import {FlButton} from 'fall-ui'

    app.use(FallUI)
